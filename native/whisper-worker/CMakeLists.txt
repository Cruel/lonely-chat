cmake_minimum_required(VERSION 3.16)
project(lonely_whisper_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
if(NOT DEFINED WHISPER_SOURCE_DIR)
  message(FATAL_ERROR "Pass WHISPER_SOURCE_DIR for the pinned whisper.cpp source")
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper-build)
add_executable(lonely-whisper-worker main.cpp)
target_link_libraries(lonely-whisper-worker PRIVATE whisper)
